121def Sample2DTracersFromField(delta_sampling, mask, N_realisations, n_tracers, seed=None, map_NSIDE=64):
122    '''
123    Creates a number of realisations of discrete tracers sampled from a given 2D angular overdensity field using Poisson sampling.
124
125    Parameters
126    ----------
127    delta_sampling : float ndarray
128        HEALPix overdensity field from which tracers are to be sampled.
129
130    mask : int ndarray
131        HEALPix mask defining the survey footprint. Pixels within the footprint should have value 1, while pixels outside the footprint should have value ``hp.UNSEEN``.
132
133    N_realisations : int
134        total number of realisations to be sampled.
135
136    n_tracers : int
137        total number of tracers to be sampled.
138
139    seed : int, optional
140        random seed for reproducibility, by default None.
141
142    map_NSIDE : int, optional
143        HEALPix nside of the output number counts map, by default 64.
144
145    Returns
146    -------
147    
148    tracer_pos_masked_ds_arr : float ndarray
149        array of shape (``N_realisations``, ``n_tracers``, 2) containing the sampled tracer positions in radians. The last dimension contains (Dec, RA) pairs.
150
151    map : float ndarray
152        HEALPix map of the number counts of the sampled tracers, summed over all realisations.
153    '''
154
155    #-----------------------------------------------------------------------------------------------
156
157    NSIDE = hp.npix2nside(len(delta_sampling))
158    Nobj = int(1.5*n_tracers*12*NSIDE**2/len(np.where(mask==1)[0]))
159
160    tracer_pos_arr = np.zeros((N_realisations, Nobj, 2))
161    tracer_pos_masked_ds_arr = np.zeros((N_realisations, n_tracers, 2))
162
163    np.random.seed(seed)
164
165    for i, tracer_pos in enumerate(tracer_pos_arr):
166
167        sampled_pix = hpt.rand_pix_from_map(1+delta_sampling, n=Nobj)
168        sampled_RAs, sampled_Decs = hp.pix2ang(NSIDE, sampled_pix, lonlat=True)
169        tracer_pos[:, 0] = np.deg2rad(sampled_Decs)
170        tracer_pos[:, 1] = np.deg2rad(sampled_RAs)
171        ipix = hp.ang2pix(NSIDE, np.rad2deg(tracer_pos[:, 1]), np.rad2deg(tracer_pos[:, 0]), nest=False, lonlat=True)
172        mask_val = mask[ipix]
173        ind_masked = np.where(mask_val!=hp.UNSEEN)[0]
174        tracer_pos_masked = tracer_pos[ind_masked]
175        # print(len(tracer_pos_masked), n_tracers)
176        tracer_pos_masked_ds_arr[i] = tracer_pos_masked[np.random.choice(len(tracer_pos_masked), n_tracers, replace=False)]
177
178    map = cat2hpx(np.rad2deg(tracer_pos_masked_ds_arr)[:, :, 1], np.rad2deg(tracer_pos_masked_ds_arr)[:, :, 0], nside=map_NSIDE, radec=False).astype(float)
179
180    return tracer_pos_masked_ds_arr, map

184def Sample2DPoissonTracers(mask, N_realisations, n_tracers, seed=None, map_NSIDE=64):
185    '''
186    Creates a number of realisations of discrete tracers sampled from a uniform Poisson distribution over the sky.
187
188    Parameters
189    ----------
190    mask : int ndarray
191        HEALPix mask defining the survey footprint. Pixels within the footprint should have value 1, while pixels outside the footprint should have value ``hp.UNSEEN``.
192
193    N_realisations : int
194        total number of realisations to be sampled.
195
196    n_tracers : int
197        total number of tracers to be sampled.
198
199    seed : int, optional
200        random seed for reproducibility, by default None.
201
202    map_NSIDE : int, optional
203        HEALPix nside of the output number counts map, by default 64.
204
205    Returns
206    -------
207    
208    randoms_pos_masked_ds_arr : float ndarray
209        array of shape (``N_realisations``, ``n_tracers``, 2) containing the sampled tracer positions in radians. The last dimension contains (Dec, RA) pairs.
210
211    map : float ndarray
212        HEALPix map of the number counts of the sampled tracers, summed over all realisations.
213    '''
214
215    #-----------------------------------------------------------------------------------------------
216
217    NSIDE = hp.npix2nside(len(mask))
218    Nobj = int(1.5*n_tracers*12*NSIDE**2/len(np.where(mask==1)[0]))
219
220    randoms_pos_arr = np.zeros((N_realisations, Nobj, 2))
221    randoms_pos_masked_ds_arr = np.zeros((N_realisations, n_tracers, 2))
222
223    np.random.seed(seed)
224
225    for i, randoms_pos in enumerate(randoms_pos_arr):
226
227        randoms_pos[:, 0] = np.arcsin(np.random.uniform(low=-1, high=1, size=randoms_pos.shape[0]))
228        randoms_pos[:, 1] = np.random.uniform(low=0, high=2*np.pi, size=randoms_pos.shape[0])
229        ipix = hp.ang2pix(NSIDE, np.rad2deg(randoms_pos[:, 1]), np.rad2deg(randoms_pos[:, 0]), nest=False, lonlat=True)
230        mask_val = mask[ipix]
231        ind_masked = np.where(mask_val!=hp.UNSEEN)[0]
232        randoms_pos_masked = randoms_pos[ind_masked]
233        randoms_pos_masked_ds_arr[i] = randoms_pos_masked[np.random.choice(len(randoms_pos_masked), n_tracers, replace=False)]
234
235    map = cat2hpx(np.rad2deg(randoms_pos_masked_ds_arr)[:, :, 1], np.rad2deg(randoms_pos_masked_ds_arr)[:, :, 0], nside=map_NSIDE, radec=False).astype(float)
236
237    return randoms_pos_masked_ds_arr, map

241def Sample3DPoissonTracers(N_realisations, n_tracers, boxsize=1000, starting_seed=42):
242    '''
243    Creates a number of realisations of discrete tracers sampled from a uniform Poisson distribution in 3D space.
244
245    Parameters
246    ----------
247    N_realisations : int
248        total number of realisations to be sampled.
249
250    n_tracers : int
251        total number of tracers to be sampled.
252
253    boxsize : float, optional
254        size of the cubic box in which the tracers are to be sampled, by default 1000.0 Mpc/h.
255
256    starting_seed : int, optional
257        random seed for reproducibility, by default 42. This is the seed set for the first realisation,
258        Henceforth, the next ith realisation has seed = starting_seed + i.
259
260    Returns
261    -------
262    
263    randoms_pos_ds_arr : float ndarray
264        array of shape (``N_realisations``, ``n_tracers``, 3) containing the sampled tracer positions in Mpc/h.
265    '''
266
267    #-----------------------------------------------------------------------------------------------
268
269    randoms_pos_ds_arr = []
270
271    np.random.seed(starting_seed)
272
273    for i in range(N_realisations):
274
275        randoms_pos_ds_arr.append(np.random.uniform(low=0, high=boxsize, size=(n_tracers, 3)))
276    
277    randoms_pos_ds_arr = np.array(randoms_pos_ds_arr)
278
279    return randoms_pos_ds_arr

283def Sample2DTracersFromQuijoteBox(sim_num, tracer_type, mask, N_realisations, n_tracers, seed=None, map_NSIDE=64, DataPath='../kNNpy/Data'):
284    '''
285    Creates a number of realisations of discrete tracers sampled from dark-matter halo catalogues created using the Quijote simulation hight-resolution boxes[^1]. The 3D tracer positions are converted to angular 2D coordinates by projecting them onto the sky of a hypothetical observer at the centre of the simulation box. Currently supports two types of tracers, namely 'Galaxies' and 'Clusters' delineated based on a simple mass cut; galaxies are halos with mass ``M_halo >= 3e13 Msun/h and M_halo <= 1e14 Msun/h``, while clusters are halos with mass ``M_halo >= 1e14 Msun/h and M_halo <= 2.5e14 Msun/h``.
286
287    Parameters
288    ----------
289    sim_num : int
290        Quijote simulation realisation number to be used for sampling the tracers.
291
292    tracer_type : str
293        type of tracers to be sampled. Currently supports 'Galaxies' and 'Clusters'.
294
295    mask : int ndarray
296        HEALPix mask defining the survey footprint. Pixels within the footprint should have value 1, while pixels outside the footprint should have value ``hp.UNSEEN``.
297
298    N_realisations : int
299        total number of realisations to be sampled.
300
301    n_tracers : int
302        total number of tracers to be sampled. Must be less than the total number of tracers available in the given simulation box after projecting onto the virtual sky and applying the survey mask. If the observational footprint covers the entire sky, the maximum number of 'Clusters' ('Galaxies') that can be sampled is <enter limit here> (<enter limit here>). For partial sky coverage, this number might be significantly lower.
303
304    seed : int, optional
305        random seed for reproducibility, by default None.
306
307    map_NSIDE : int, optional
308        HEALPix nside of the output number counts map, by default 64.
309
310    DataPath : str, optional
311        path to the kNNpy Data directory, by default '../kNNpy/Data/'.
312
313    Returns
314    -------
315    
316    tracer_pos_masked_ds_arr : float ndarray
317        array of shape (``N_realisations``, ``n_tracers``, 2) containing the sampled tracer positions in radians. The last dimension contains (Dec, RA) pairs.
318
319    map : float ndarray
320        HEALPix map of the number counts of the sampled tracers, summed over all realisations.
321
322    References
323    ----------
324    [^1]: <add reference here>
325    '''
326
327    #-----------------------------------------------------------------------------------------------
328
329    #Defining the quijote box size
330    boxsize=1000.0  #Mpc/h
331
332    #Converting the simulation number into a simulation name string. For example, when sim_num is 5, name will be 'sim_005' and when sim_num is 50, name will be 'sim_050'
333    num = str(sim_num).zfill(3)
334
335    #Getting the NSIDE from the mask
336    NSIDE = hp.npix2nside(len(mask))
337    Nobj = int(1.5*n_tracers*12*NSIDE**2/len(np.where(mask==1)[0]))
338
339    if tracer_type=='Galaxies': 
340        
341        #Reading in the halo catalogs corresponding to the given simulation number
342        folder_ga = f'{DataPath}/Quijote_simulations/Fiducial_HR_rockstar/Galaxy_cat'
343        filepath_ga = '{a}/sim_{b}'.format(a = folder_ga, b = num)
344        ga_data_selected = pd.read_csv(filepath_ga)
345
346        #Positions of halos
347        all_ga_pos = np.zeros((len(ga_data_selected), 3), dtype = np.float32)
348        all_ga_pos[:, 0] = ga_data_selected.loc[:, '8'].values
349        all_ga_pos[:, 1] = ga_data_selected.loc[:, '9'].values
350        all_ga_pos[:, 2] = ga_data_selected.loc[:, '10'].values
351
352        #Converting 3D positions to sky angles, observer at centre of box
353        all_angpos = cartesian_corner_to_angles_centre(all_ga_pos[:, 0], all_ga_pos[:, 1], all_ga_pos[:, 2], boxsize, boxsize/2)
354
355
356    elif tracer_type=='Clusters':
357        
358        #Reading in the halo catalogs corresponding to the given simulation number
359        folder_cl = f'{DataPath}/Quijote_simulations/Fiducial_HR_rockstar/Cluster_cat'
360        filepath_cl = '{a}/sim_{b}'.format(a = folder_cl, b = num)
361        cl_data_selected = pd.read_csv(filepath_cl)
362
363        #Positions of halos
364        all_cl_pos = np.zeros((len(cl_data_selected), 3), dtype = np.float32)
365        all_cl_pos[:, 0] = cl_data_selected.loc[:, '8'].values
366        all_cl_pos[:, 1] = cl_data_selected.loc[:, '9'].values
367        all_cl_pos[:, 2] = cl_data_selected.loc[:, '10'].values
368
369        #Converting 3D positions to sky angles, observer at centre of box
370        all_angpos = cartesian_corner_to_angles_centre(all_cl_pos[:, 0], all_cl_pos[:, 1], all_cl_pos[:, 2], boxsize, boxsize/2)
371
372    else:
373        raise ValueError("Invalid tracer type specified. Supported types are 'Galaxies' and 'Clusters'.")
374    
375    tracer_pos_arr = np.zeros((N_realisations, Nobj, 2))
376    tracer_pos_masked_ds_arr = np.zeros((N_realisations, n_tracers, 2))
377
378    np.random.seed(seed)
379
380    for i, tracer_pos in enumerate(tracer_pos_arr):
381
382        sel_ind = np.random.choice(len(all_angpos), Nobj, replace=False)
383        tracer_pos = all_angpos[sel_ind]
384        ipix = hp.ang2pix(NSIDE, np.rad2deg(tracer_pos[:, 1]), np.rad2deg(tracer_pos[:, 0]), nest=False, lonlat=True)
385        mask_val = mask[ipix]
386        ind_masked = np.where(mask_val!=hp.UNSEEN)[0]
387        tracer_pos_masked = tracer_pos[ind_masked]
388        tracer_pos_masked_ds_arr[i] = tracer_pos_masked[np.random.choice(len(tracer_pos_masked), n_tracers, replace=False)]
389
390    map = cat2hpx(np.rad2deg(tracer_pos_masked_ds_arr)[:, :, 1], np.rad2deg(tracer_pos_masked_ds_arr)[:, :, 0], nside=map_NSIDE, radec=False).astype(float)
391
392    return tracer_pos_masked_ds_arr, map

397    '''
398    Samples 3D tracer particles or halos from Quijote simulation snapshots.
399    For now it only supports one tracer_type: 'particles'. Other types like 'halos', can be included if and when necessary.
400    
401    Parameters
402    ----------
403    tracer_type : str
404        type of tracers to sample ('particles' or 'halos').
405
406    N_realisations : int
407        total number of realisations to be sampled.
408
409    n_tracers : int
410        total number of tracers to be sampled.
411
412    ptype : list of int, optional, default=[1]

414

441def make_overdensity_3D(N_realisations, grid, ptype, do_RSD=False, MAS='CIC', axis=0, verbose=False, DataPath='../kNNpy/Data'):
442    '''
443    Constructs 3D overdensity fields from Gadget simulation snapshots using Mass Assignment Schemes (MAS).
444
445    Parameters
446    ----------
447    N_realisations : int
448        total number of realisations/snapshots to process.
449
450    grid : int
451        number of grid points along each axis for the 3D density field.
452
453    ptype : list of int
454        gadget particle type list to construct overdensity field.
455
456    do_RSD : bool, optional, default=False
457        whether to apply Redshift-Space Distortions (RSD).
458
459    MAS : str, optional, default='CIC'
460        Mass Assignment Scheme to use (e.g. 'NGP', 'CIC', 'TSC').
461
462    axis : int, optional, default=0
463        line-of-sight axis for RSD.
464
465    verbose : bool, optional, default=False
466        whether to print progress messages.
467
468    DataPath : str, optional, default='../kNNpy/Data'
469        path to the kNNpy Data directory.
470
471    Returns
472    -------
473    overdensity_list : float ndarray
474        array of shape (``N_realisations``, ``grid``, ``grid``, ``grid``) containing 3D overdensity fields.
475    '''
476    overdensity_list=[]
477    for i in range(N_realisations):
478        snapshot = f'{DataPath}/fiducial_LR/{i}/snapdir_004/snap_004'
479        delta = MASL.density_field_gadget(snapshot, ptype, grid, MAS, do_RSD, axis, verbose)
480        delta /= np.mean(delta, dtype=np.float64);  delta -= 1.0
481        overdensity_list.append(delta)
482    overdensity_list=np.array(overdensity_list)
483    return overdensity_list
